```rust
pub fn lookup(&self, name: &SmolStr) -> Option<&Symbol> {
    let key = SmolStr::new(name.to_ascii_uppercase());
    let mut scope_idx = self.current_scope;
    loop {
        if let Some(syms) = self.scopes[scope_idx].symbols.get(&key) {
            if let Some(first) = syms.first() {
                return Some(first);
            }
        }
        match self.scopes[scope_idx].parent {
            Some(parent) => scope_idx = parent,
            None => return None,
        }
    }
}
```

3. `lookup_qualified()` の候補収集も大文字正規化:

```rust
pub fn lookup_qualified(&self, name: &SmolStr, qualifiers: &[SmolStr]) -> Option<&Symbol> {
    if qualifiers.is_empty() {
        return self.lookup(name);
    }
    let key = SmolStr::new(name.to_ascii_uppercase());
    let candidates: Vec<&Symbol> = self
        .scopes
        .iter()
        .filter_map(|scope| scope.symbols.get(&key))
        .flat_map(|syms| syms.iter())
        .collect();
    candidates
        .into_iter()
        .find(|candidate| self.matches_qualifiers(candidate, qualifiers))
}
```

```rust
fn find_all_symbols(&self, name: &SmolStr) -> Vec<&Symbol> {
    let key = SmolStr::new(name.to_ascii_uppercase());
    let mut result: Vec<&Symbol> = Vec::new();
    for scope in &self.scopes {
        if let Some(syms) = scope.symbols.get(&key) {
            result.extend(syms.iter());
```

### Task 2: codegen の find_data_item をHashMapキャッシュ化

**Files:**
- Modify: `crates/cobol-codegen/src/codegen.rs:6474-6501, 7909-7975`

**問題:** `find_data_item()` と `find_data_item_size_in()` が再帰的線形探索（O(n) × 呼び出し回数）。codegen全体で36回以上呼ばれ、ループ内からの呼び出しもあるため O(n × m)。

**Step 1: キャッシュ用の構造体と構築関数を追加**

`codegen.rs` 冒頭付近に追加:

```rust
/// Pre-built lookup table for data items by sanitized C name.
/// Avoids O(n) recursive search on every find_data_item call.
struct DataItemIndex {
    /// Maps sanitized C name -> (byte_size, HirType clone)
    size_map: HashMap<String, u32>,
}

impl DataItemIndex {
    fn build(items: &[HirDataItem]) -> Self {
        let mut size_map = HashMap::new();
        Self::index_items(items, &mut size_map);
        DataItemIndex { size_map }
    }

    fn index_items(items: &[HirDataItem], size_map: &mut HashMap<String, u32>) {
        for item in items {
            let c_name = sanitize_name(&item.name);
            let size = data_item_byte_size(&item.data_type);
            size_map.entry(c_name).or_insert(size);
            if let HirType::Group { members, .. } = &item.data_type {
                Self::index_items(members, size_map);
            }
        }
    }

    fn get_size(&self, c_name: &str) -> u32 {
        // Handle subscript/struct-access expressions
        if c_name.contains('[') || c_name.contains(".members.") {
            if let Some(pos) = c_name.rfind(".members._m_") {
                let leaf = &c_name[pos + ".members._m_".len()..];
                let leaf_name = if let Some(br) = leaf.find('[') {
                    &leaf[..br]
                } else {
                    leaf
                };
                if !leaf_name.is_empty() {
                    if let Some(&size) = self.size_map.get(leaf_name) {
                        if size > 0 {
                            return size;
                        }
                    }
                }
            }
        }
        self.size_map.get(c_name).copied().unwrap_or(0)
    }
}
```

**Step 2: generate_c_code() でインデックスを構築し、find_data_item_size の呼び出しをインデックス経由に変更**

`generate_c_code()` 関数の冒頭で `DataItemIndex::build(&program.data_items)` を呼び、`find_data_item_size()` の代わりに `index.get_size()` を使うように置換。

変更点が多いため、段階的に:
1. まず `DataItemIndex` を構築するコードを `generate_c_code()` に追加
2. `find_data_item_size()` の中身を `DataItemIndex::get_size()` のロジックに切り替え（引数に `&DataItemIndex` を追加するか、thread_local で渡す）
3. 最もシンプルな方法: `find_data_item_size()` の内部で使う `find_data_item_size_in()` の再帰探索をそのまま残しつつ、`generate_c_code()` のスコープで `RefCell<HashMap>` キャッシュを使う

実装方針: `find_data_item_size()` にキャッシュ引数を追加するのは呼び出し箇所が多すぎるため、`thread_local!` で `RefCell<HashMap<String, u32>>` キャッシュを使う。

```rust
thread_local! {
    static DATA_ITEM_SIZE_CACHE: RefCell<HashMap<String, u32>> = RefCell::new(HashMap::new());
}

fn init_data_item_size_cache(items: &[HirDataItem]) {
    DATA_ITEM_SIZE_CACHE.with(|cache| {
        let mut map = cache.borrow_mut();
        map.clear();
        populate_size_cache(items, &mut map);
    });
}

fn populate_size_cache(items: &[HirDataItem], map: &mut HashMap<String, u32>) {
    for item in items {
        let c_name = sanitize_name(&item.name);
        let size = data_item_byte_size(&item.data_type);
        map.entry(c_name).or_insert(size);
        if let HirType::Group { members, .. } = &item.data_type {
            populate_size_cache(members, map);
        }
    }
}
```

`find_data_item_size()` を修正してキャッシュを先にチェック:

```rust
fn find_data_item_size(c_name: &str, data_items: &[HirDataItem]) -> u32 {
    // Handle subscript/struct-access - extract leaf name
    let lookup_name = if c_name.contains('[') || c_name.contains(".members.") {
        if let Some(pos) = c_name.rfind(".members._m_") {
            let leaf = &c_name[pos + ".members._m_".len()..];
            let leaf_name = if let Some(br) = leaf.find('[') {
                &leaf[..br]
            } else {
                leaf
            };
            if !leaf_name.is_empty() {
                leaf_name
            } else {
                c_name
            }
        } else {
            c_name
        }
    } else {
        c_name
    };

    DATA_ITEM_SIZE_CACHE.with(|cache| {
        let map = cache.borrow();
        if let Some(&size) = map.get(lookup_name) {
            return size;
        }
        drop(map);
        // Fallback to recursive search
        find_data_item_size_in(lookup_name, data_items)
    })
}
```

```rust
fn collect_duplicate_member_names(
    items: &[HirDataItem],
    group_member_names: &BTreeSet<String>,
) -> BTreeSet<String> {
    // Remove: let group_member_names = collect_group_member_names(items);
    let mut seen = BTreeSet::new();
    let mut duplicates = BTreeSet::new();
    for item in items {
        let c_name = sanitize_name(&item.name);
        if group_member_names.contains(&c_name) {
            continue;
        }
        if let HirType::Group { members, .. } = &item.data_type {
            let mut local_names = BTreeSet::new();
            collect_member_names_recursive(members, &mut local_names);

                    duplicates.insert(name.clone());
                }
            }
        }
    }
    duplicates
}
```
